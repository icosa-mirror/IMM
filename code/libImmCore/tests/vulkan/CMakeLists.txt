cmake_minimum_required(VERSION 3.20)
project(ImmVulkanHostLifetime LANGUAGES CXX)
add_executable(imm_vulkan_host_frame_slots host_frame_slots.cpp)
target_compile_features(imm_vulkan_host_frame_slots PRIVATE cxx_std_11)
if(MSVC)
    target_compile_options(imm_vulkan_host_frame_slots PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(imm_vulkan_host_frame_slots PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME vulkan_host_frame_lifetime COMMAND imm_vulkan_host_frame_slots)
